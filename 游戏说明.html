一、项目总览

这是什么：一个竖版射击游戏。玩家控制底部飞机，发射子弹消灭上方落下的敌机。三种敌机（小/中/大），爆炸粒子特效，难度逐渐增加，星空背景滚动。

窗口 720×960，深空背景 + 视差星星

关键技术说明
状态机MENU → PLAYING → GAMEOVER 三态切换，int 变量控制
对象池预分配数组 + active 标记，全程无 malloc/free
AABB 碰撞轴对齐矩形碰撞检测，X/Y 投影分离判断
粒子系统随机速度 + 重力 + 颜色渐变，模拟爆炸火花
双缓冲BeginBatchDraw / FlushBatchDraw 防画面闪烁

三、五种结构体（数据模型）

设计思想：每种游戏实体对应一个结构体。用 active 字段管理生命周期（1=活跃/0=空闲），实现对象池——预分配数组，循环复用，零内存分配。

① 玩家飞机

typedef struct {
    float x, y;               // 坐标（float 保证平滑移动）
    int   w, h;               // 碰撞体积（宽×高）
    int   lives;              // 剩余命数
    int   invTimer;           // 无敌倒计时（>0=无敌中）
    int   shootTimer;         // 射击冷却倒计时（>0=不能射）
} Player;

② 子弹

typedef struct {
    float x, y;  int w, h;
    int   active;  // ⭐ 对象池核心：1=使用中, 0=空闲可复用
} Bullet;

③ 敌机

typedef struct {
    float x, y;  int w, h;
    int   speed;   // 下落速度
    int   hp;      // 血量：小1/中2/大4
    int   type;    // 0=小 1=中 2=大
    int   active;
} Enemy;

④ 爆炸粒子

typedef struct {
    float x, y;              // 当前位置
    float vx, vy;            // 速度向量（方向+快慢）
    int   life, maxLife;     // 剩余/总寿命（帧），ratio=life/maxLife 控制颜色
    int   active;
} Particle;

⑤ 背景星星

typedef struct {
    float x, y;
    int   speed;    // 1(远/暗/慢) ~ 3(近/亮/快) → 视差效果
    int   bright;   // 亮度 100~255
} Star;

💡 答辩话术

"五种实体各一个结构体，用 active 标记管理生命周期。这就是轻量级的对象池模式——预分配固定数组，用的时候激活，不用了回收。全程没有一个 malloc，也没有一个 free。"

⭐ 对象池的三个核心函数

Bullet* getFreeBullet() {
    for(int i=0; i<MAX_BULLETS; i++)
        if(!g_bullets[i].active)        // 找空闲槽位
            return &g_bullets[i];        // 返回地址 → 填入新数据 → 激活
    return NULL;  // 池满（正常游戏不可能满）
}
// getFreeEnemy() 和 getFreeParticle() 逻辑完全一样

🔄 对象池生命周期

  数组初始化（active 全 = 0）
        │
        ▼
  ┌─ 需要新对象 ──→ getFreeXXX() 遍历找 active==0 的槽
  │                      │
  │                      ▼
  │               填入数据（x,y,w,h,hp...）
  │               设置 active = 1
  │                      │
  │                      ▼
  │               对象在游戏中活动（飞、碰撞、渲染）
  │                      │
  │                      ▼
  │               条件触发（飞出屏幕 / 死亡 / 寿命到）
  │                      │
  │                      ▼
  │               设置 active = 0 ←── 回收！
  │                      │
  └──────────────────────┘

五、实体生成

生成子弹 — 双发

void spawnBullet(){
    Bullet*b=getFreeBullet(); if(!b)return;
    b->x=g_player.x;                         // 从玩家 X 坐标
    b->y=g_player.y - g_player.h/2;         // 从机头射出
    b->w=6; b->h=14; b->active=1;
}
// 实际调用时左右翼各发一颗 → 双发效果

生成敌机 — 随机类型

void spawnEnemy(){
    Enemy*e=getFreeEnemy(); if(!e)return;
    int r=rand()%100;
    if(r<70){      // 70% → 小敌机（快、脆、10分）
        e->type=ENEMY_SMALL; e->w=30; e->speed=rand()%3+3; e->hp=1;
    }else if(r<90){ // 20% → 中敌机
        e->type=ENEMY_MEDIUM; e->w=40; e->speed=rand()%2+2; e->hp=2;
    }else{           // 10% → 大敌机（慢、肉、左右摇摆、50分）
        e->type=ENEMY_BIG; e->w=55; e->speed=1; e->hp=4;
    }
    e->x=(float)(rand()%(WINDOW_WIDTH-e->w)+e->w/2); // 随机X不超边界
    e->y=(float)(-e->h); // 从屏幕顶部外飞入
    e->active=1;
}

⭐ 生成爆炸粒子 — 答辩重点

void spawnExplosion(float x,float y,int count){
    for(int i=0; i<count; i++){
        Particle*p=getFreeParticle(); if(!p)break;
        p->x=x; p->y=y;
        // 随机方向 0~360°（转弧度）
        float angle=(float)(rand()%360)*3.14159f/180.0f;
        float speed=(float)(rand()%5)+2.0f;
        p->vx=cosf(angle)*speed;  // X 分量 = cos(角度)×速度
        p->vy=sinf(angle)*speed;  // Y 分量 = sin(角度)×速度
        p->maxLife=rand()%20+15;   // 存活 15~35 帧
        p->life=p->maxLife; p->active=1;
    }
}

三角函数的作用：angle 是发射角度（0~360°），cos(angle) 和 sin(angle) 把角度拆成 X 方向分量和 Y 方向分量，乘上速度就是向该方向飞。360 个粒子均匀散布 = 圆形爆炸。

六、更新逻辑（核心帧循环）

updatePlayer() — 玩家

做的事代码逻辑
移动检测 WASD/方向键 → 加减 x/y → imax/imin 限制不超出屏幕边界
射击冷却shootTimer 每帧 -1，到 0 才能发射
发射子弹空格按下 + 冷却完毕 → spawnBullet() ×2（左右双发）→ shootTimer=8
无敌倒计时invTimer 每帧 -1，>0 时碰撞检测跳过

updateBullets() — 子弹

for(int i=0; i<MAX_BULLETS; i++){
    if(!g_bullets[i].active) continue;   // 跳过不活跃的
    g_bullets[i].y -= BULLET_SPEED;         // 往上飞
    if(g_bullets[i].y < -20)               // 飞出屏幕顶
        g_bullets[i].active = 0;           // 回收
}

逻辑最简单——飞就完了，出屏幕就回收。

updateEnemies() — 敌机

for(int i=0; i<MAX_ENEMIES; i++){
    if(!g_enemies[i].active) continue;
    g_enemies[i].y += (float)g_enemies[i].speed;  // 下落
    if(g_enemies[i].y > WINDOW_HEIGHT+60)    // 飞出底部
        g_enemies[i].active=0;                    // 回收

    // 大敌机左右摇摆（sin 产生 -1~1 波动）
    if(g_enemies[i].type == ENEMY_BIG)
        g_enemies[i].x += sinf((float)g_frameCount*0.05f+(float)i)*1.5f;
}

// === 难度曲线：敌机生成加速 ===
g_enemySpawnTimer++;
if(g_enemySpawnTimer >= g_enemySpawnRate){
    g_enemySpawnTimer=0; spawnEnemy();
    g_enemySpawnRate = imax(15, 45 - g_score/500);
    // ↑ 分数每多500，生成间隔减1帧（最快15帧）
}

updateParticles() — 粒子

for(int i=0; i<MAX_PARTICLES; i++){
    if(!g_particles[i].active) continue;
    g_particles[i].x  += g_particles[i].vx;   // 沿速度方向移动
    g_particles[i].y  += g_particles[i].vy;
    g_particles[i].vy += 0.15f;              // 重力：速度向下偏转
    g_particles[i].life--;                    // 寿命-1
    if(g_particles[i].life <= 0)
        g_particles[i].active = 0;          // 寿终正寝
}

七、⭐ AABB 碰撞检测（答辩核心）

AABB = Axis-Aligned Bounding Box（轴对齐包围盒）

        物体1                      物体2
    l1 ┌────────┐ r1           l2 ┌──────┐ r2
       │  (x1,y1) │               │(x2,y2)│
    t1 │   w1×h1 │ b1         t2 │ w2×h2 │ b2
       └────────┘               └──────┘

    X轴投影重叠？  l1 < r2  且  r1 > l2
    Y轴投影重叠？  t1 < b2  且  b1 > t2

    两个轴都重叠 → 碰撞！

int checkCollision(float x1,float y1,int w1,int h1,
                   float x2,float y2,int w2,int h2){
    // 物体1的四条边界 = 中心坐标 ± 半宽/半高
    float l1 = x1 - w1/2.0f, r1 = x1 + w1/2.0f;  // 左、右
    float t1 = y1 - h1/2.0f, b1 = y1 + h1/2.0f;  // 上、下

    // 物体2的四条边界
    float l2 = x2 - w2/2.0f, r2 = x2 + w2/2.0f;
    float t2 = y2 - h2/2.0f, b2 = y2 + h2/2.0f;

    // 四个条件同时成立 → 碰撞
    return (l1 < r2 && r1 > l2 &&    // X轴重叠
            t1 < b2 && b1 > t2);      // Y轴重叠
}

💡 答辩用一句话

"AABB 碰撞检测——分别判断 X 轴和 Y 轴的投影是否重叠。如果任意一个轴的投影不重叠，两个矩形就一定不碰撞。这其实是分离轴定理在轴对齐情况下的简化版。"

碰撞检测的两个应用场景

场景做了什么特殊处理
子弹 vs 敌机双重循环遍历所有子弹×敌机。命中→子弹消失、敌机扣血/死亡爆炸一颗子弹只打一架（break）
玩家 vs 敌机遍历所有敌机。碰撞→敌机炸、玩家扣命、进入无敌无敌中直接跳过检测

八、渲染系统

双缓冲原理

模式工作方式效果
单缓冲 ❌直接往屏幕画 → 每笔都能看到画面闪烁
双缓冲 ✅先画到后台 → 一帧画完整张替换完全流畅
// main() 开头
BeginBatchDraw();   // 开启双缓冲模式

// 每一帧结束时
FlushBatchDraw();  // 将后台缓冲一次性提交到屏幕

// 程序退出时
EndBatchDraw();     // 关闭双缓冲

  第6层：drawUI()            ← 分数/血量栏（最顶层）

九、粒子系统详解

一颗粒子的完整生命周期

  spawnExplosion()
    │
    ├→ 随机角度 (0°~360°) → 用三角函数拆成 vx, vy
    ├→ 随机速度 (2~6)
    ├→ 初始生命 = 15~35 帧
    │
    ▼
  每帧：
    x += vx    (水平移动)
    y += vy    (垂直移动)
    vy += 0.15 (重力拉弯轨迹)
    life -= 1
    │
    ├→ ratio = life / maxLife
    │   > 0.6 → 黄色（刚爆炸，最亮）
    │   0.3~0.6 → 橙色（中场）
    │   < 0.3 → 暗红色（快熄灭）
    │
    ▼
  life ≤ 0 → active = 0（回收）

  粒子大小也随 ratio 变化：size = 3×ratio + 1
  → 刚生成大，随时间缩小

💡 答辩话术

"爆炸时在中心生成几十个粒子，每个粒子有随机发射方向、初始速度、固定寿命。粒子受重力影响，轨迹向下弯曲。颜色从黄色渐变到橙色再到暗红色，大小也同步缩小——模拟高温火花逐渐冷却熄灭的物理过程。这和商业游戏的粒子特效原理一样，只是他们用 GPU 着色器实现，我这里是逐帧 CPU 计算。"

十、主循环 & 状态机

                    ┌──────────┐
        ┌─── SPACE →│  PLAYING │←──── R ───┐
        │           │ update() │           │
        │           │ render() │           │
        │           └────┬─────┘           │
        │              Q │                 │
        ▼                ▼                 │
   ┌─────────┐                    ┌───────────┐
   │  MENU   │                    │ GAMEOVER  │
   │ drawMenu│     玩家死亡 →     │drawGameOver│
   └─────────┘                    └───────────┘
        │                               │
        └──── ESC ────→ 退出 ←── ESC ──┘

int main(){
    initgraph(720,960);        // 打开图形窗口
    BeginBatchDraw();          // 开启双缓冲
    srand((unsigned)time(NULL)); // 随机种子 = 当前时间
    initGame();                // 初始化一切
    g_state = STATE_MENU;       // 初始状态 = 菜单

    while(1){                    // 死循环，游戏一直跑
        g_frameCount++;         // 帧号+1

        switch(g_state){        // 根据状态做不同的事

        case STATE_MENU:
            drawMenu();             // 画菜单
            if(KEY_DOWN(VK_SPACE))  // 空格 → 开始游戏
                { initGame(); g_state=STATE_PLAYING; }
            if(KEY_DOWN(VK_ESCAPE)) // ESC → 退出
                goto exitGame;
            break;

        case STATE_PLAYING:
            updateGame(); render();  // 更新+渲染
            if(KEY_DOWN('Q'))        // Q → 回菜单
                g_state = STATE_MENU;
            if(KEY_DOWN(VK_ESCAPE))
                goto exitGame;
            break;

        case STATE_GAMEOVER:
            drawGameOver();         // 画结束画面
            if(KEY_DOWN('R'))        // R → 重来
                { initGame(); g_state=STATE_PLAYING; }
            if(KEY_DOWN(VK_ESCAPE))
                goto exitGame;
            updateStars();           // 星星继续动
            updateParticles();      // 粒子继续散
            break;
        }

        Sleep(16);               // 等16ms → ≈60 FPS
    }

exitGame:
    EndBatchDraw();           // 关双缓冲
    closegraph();              // 关窗口
    return 0;
}

💡 为什么用 goto？

老师大概率会问。标准回答："exitGame 是统一的清理出口。不用 goto 的话，需要在三个 case 里都写 EndBatchDraw+closegraph+return，代码重复三遍。goto 在这里充当了清理函数的作用，比复制粘贴更不容易出错。这是 C 语言里公认的 goto 最佳实践。"

【碰撞检测 1分钟 ⭐】

"碰撞用的是 AABB——轴对齐包围盒。把每个物体简化成一个矩形，计算四条边界。分别判断 X 轴和 Y 轴的投影是否重叠。任一轴不重叠就一定不碰撞——分离轴定理的简化版。复杂度 O(n×m)，当前规模完全够用。"

【粒子系统 30秒】

"敌机死亡时生成几十个粒子，随机发射方向、带重力加速度、颜色从黄渐变到暗红——模拟火花熄灭。和商业游戏原理一样，只是在 CPU 上算。"

十二、Q&A 题库

Q1: 为什么用数组不用链表？

固定池大小可以预判内存上限。数组访问 O(1)，链表遍历 O(n)。而且游戏实体数量天然有上限，超过就不生成。

Q2: 双缓冲什么原理？

BeginBatchDraw 把所有绘制操作缓存到后台缓冲区，FlushBatchDraw 一次性替换屏幕。如果直接往屏幕上画，玩家会看到绘制过程——闪烁。

Q3: 碰撞检测精度够吗？

对本游戏足够了。两个圆形物体用矩形检测误差很小。要更精确可以用圆形碰撞（比较圆心距和半径和），但性能开销更大。

Q4: 物体会不会穿透？

当前速度下不会——子弹10像素/帧，碰撞体最小30像素。如果速度超过自身宽度确实会穿（tunneling），需要连续碰撞检测（CCD）。但本游戏没到这个阈值。

Q5: 状态机为什么用 int 不用 enum？

C 语言习惯，#define 更直观。用 enum 类型更安全——这是可以改的地方。功能上没区别。

Q6: 怎么扩展双人？

Player 改数组、第二人用方向键、碰撞检测加上玩家间碰撞。框架已经支持了。

Q7: goto 的问题？

exitGame 是统一清理出口。不用 goto 代码重复三遍。C 语言里 goto 做错误处理和资源清理是公认的最佳实践——Linux 内核也大量使用。

Q8: 难度怎么设计的？

三维度：敌机生成加速（45→15帧）、敌机类型比例（7:2:1，大敌机4血+摇摆）、无敌保护（90帧）。平滑上升不阶梯。
